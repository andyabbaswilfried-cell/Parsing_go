	var res []string
	for _,r := range phrase{
		
		if nbr> len(phrase){
			fmt.Println("il y a un problème de mots à mettre en majuscules")
		} else if nbr >=0 {
			if r == phrase[len(phrase)-nbr]{
				res = append(res,Up(phrase[len(phrase)-nbr]))
				nbr--
			}else if nbr==0{
				return  res
			} else {

		}
	}

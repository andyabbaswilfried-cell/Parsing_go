func Ponctuation_simple(phrase []string) []string {
	var res []string

	for i := 0; i < len(phrase); i++ {
		if phrase[i] == "." || phrase[i] == "!" || phrase[i] == ":" {
			res[len(res)-1] = res[len(res)-1] + phrase[i] + " "

		} else if phrase[i] == "..." || phrase[i] == "!!" || phrase[i] == "!?" {
			res[len(res)-1] = res[len(res)-1] + phrase[i] + " "
			i++
		} else {
			res = append(res, phrase[i])
		}
	}
	return res
}
